cmake_minimum_required(VERSION 3.24)
project(managed_vdd_poc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SUNSHINE_ROOT "../.." ABSOLUTE)
option(VDD_BUILD_WINDOWS_PROVIDER "Build the Windows device probe and provider" ON)
enable_testing()
add_library(managed_vdd_core "${SUNSHINE_ROOT}/src/managed_vdd.cpp" "${SUNSHINE_ROOT}/src/vdd_modes.cpp")
target_include_directories(managed_vdd_core PUBLIC "${SUNSHINE_ROOT}")
add_subdirectory("${SUNSHINE_ROOT}/third-party/lizardbyte-common/third-party/googletest" googletest EXCLUDE_FROM_ALL)
add_executable(test_sunshine "${SUNSHINE_ROOT}/tests/unit/test_managed_vdd.cpp" "${SUNSHINE_ROOT}/tests/unit/test_vdd_modes.cpp")
set_target_properties(test_sunshine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests")
target_link_libraries(test_sunshine PRIVATE managed_vdd_core GTest::gtest_main)
add_test(NAME managed_vdd COMMAND test_sunshine)
if(WIN32 AND VDD_BUILD_WINDOWS_PROVIDER)
    add_subdirectory("${SUNSHINE_ROOT}/third-party/libdisplaydevice" libdisplaydevice)
    add_library(managed_vdd_windows "${SUNSHINE_ROOT}/src/platform/windows/managed_vdd.cpp")
    target_compile_definitions(managed_vdd_windows PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(managed_vdd_windows PUBLIC managed_vdd_core libdisplaydevice::display_device nlohmann_json::nlohmann_json setupapi cfgmgr32 ole32 uuid)
    add_executable(managed-vdd-poc main.cpp)
    target_link_libraries(managed-vdd-poc PRIVATE managed_vdd_windows)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_link_options(managed-vdd-poc PRIVATE -static)
    endif()
    target_sources(test_sunshine PRIVATE "${SUNSHINE_ROOT}/tests/unit/test_managed_vdd_windows.cpp")
    target_link_libraries(test_sunshine PRIVATE managed_vdd_windows)
endif()
